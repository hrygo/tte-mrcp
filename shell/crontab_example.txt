# Crontab configuration for unimrcp server monitoring
# Edit crontab: crontab -e
# View crontab: crontab -l

# ============================================
# 生产环境配置

# Check unimrcpserver status every minute
# If process is down, it will be auto-started

# Clean logs older than 7 days, run daily at 2:00 AM
0 2 * * * ${HOME}/shell/clean_logs.sh >/dev/null 2>&1
